package mypack;

import java.sql.*;

public class Q5_a {
	private static final String URL="jdbc:mysql://localhost:3306/dbName",USER="root",PASS="";
	
	public static void main(String[] args) {
		try {
			Connection conn=DriverManager.getConnection(URL,USER,PASS);
			Statement stmt=conn.createStatement(ResultSet.TYPE_SCROLL_INSENSITIVE,ResultSet.CONCUR_UPDATABLE);
			
			String createDB="create database books";
			stmt.executeUpdate(createDB);
			System.out.print("created DB");
			
			String createTableSQL = "CREATE TABLE IF NOT EXISTS Subject (" +
                    "Code VARCHAR(10) PRIMARY KEY, " +
                    "Name VARCHAR(100), " +
                    "Department VARCHAR(100), " +
                    "Credits INT)";
            stmt.executeUpdate(createTableSQL);
            System.out.println("Table 'Subject' created or already exists.");
            
            String insertDataSQL = "INSERT INTO Subject (Code, Name, Department, Credits) VALUES " +
                    "('CS50', 'System Programming', 'CSE', 3), " +
                    "('CSL57', 'Database Sys Lab', 'CSE', 1), " +
                    "('CSL56', 'Java  Programming  Lab', 'CSE', 1), " +
                    "('HS59', 'Environmental Studies', 'Civil', 0)";
            stmt.executeUpdate(insertDataSQL);
            System.out.println("Sample data inserted into 'Subject' table.");
            
            //i. update "Java prog" to "Adv Java prog";
            String query1="update subject set Name='Advance Java programming lab' where Code='CSL57'";
            int updatedRow=stmt.executeUpdate(query1);
            System.out.println("updated table");
            
            String queryAll="select * from subject";
            ResultSet rsAll=stmt.executeQuery(queryAll);
            while (rsAll.next()) {
                System.out.println(rsAll.getString("Code") + "\t" +
                        rsAll.getString("Name") + "\t" +
                        rsAll.getString("Department") + "\t\t" +
                        rsAll.getInt("Credits"));
            }
            
            //i. delete row using prep stmt
            String query2="delete from subject where Name=?";
            PreparedStatement pstmt=conn.prepareStatement(query2);
            pstmt.setString(1, "System Programming");
            int rowsDel=pstmt.executeUpdate();
            System.out.println("deleted row");
            
            //iii. display all details
            String queryAllSubjects = "SELECT * FROM Subject";
            ResultSet rs = stmt.executeQuery(queryAllSubjects);
            System.out.println("\nCode\tName\t\t\tDepartment\tCredits");
            while (rs.next()) {
                System.out.println(rs.getString("Code") + "\t" +
                        rs.getString("Name") + "\t" +
                        rs.getString("Department") + "\t\t" +
                        rs.getInt("Credits"));
            }
			
		}catch(SQLException e) {
			e.printStackTrace();
