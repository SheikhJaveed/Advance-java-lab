package mypack;

import java.sql.*;
import java.util.*;
public class Q2_a {
	private static final String URL="jdbc:mysql://localhost:3306/dbname",USER="root",PASS="";
	
	public static void main(String[] args) {
		try {
			Connection conn=DriverManager.getConnection(URL,USER,PASS);
			Statement stmt = conn.createStatement(ResultSet.TYPE_SCROLL_INSENSITIVE, ResultSet.CONCUR_UPDATABLE);
			Scanner sc=new Scanner(System.in);
            
			String createDB="create database movies";
			stmt.executeUpdate(createDB);
			System.out.println("DB created");
			
			String createTableSQL = "CREATE TABLE IF NOT EXISTS Movies (" +
                    "ID INT PRIMARY KEY, " +
                    "Movie_Name VARCHAR(100), " +
                    "Genre VARCHAR(50), " +
                    "IMDB_Rating FLOAT, " +
                    "Year INT)";
            stmt.executeUpdate(createTableSQL);
            System.out.println("Table 'Movies' created or already exists.");
            
            String insertDataSQL = "INSERT INTO Movies (ID, Movie_Name, Genre, IMDB_Rating, Year) VALUES " +
                    "(1, 'Inception', 'Sci-fi', 8.8, 2010), " +
                    "(2, 'Orphan   ', 'Mystery', 4.9, 2009), " +
                    "(3, 'Interstellar', 'Sci-fi', 8.6, 2014), " +
                    "(4, 'La La Land', 'Musical', 8.0, 2016), " +
                    "(10, 'Gifted   ', 'Drama', 7.6, 2017)";
            stmt.executeUpdate(insertDataSQL);
            System.out.println("Sample data inserted into 'Movies' table.");

			//i.display all details
			String queryAll="select * from movies";
			ResultSet rsAll=stmt.executeQuery(queryAll);
			System.out.println("Movies table");
			printResult(rsAll);
			
			//ii. display details of 5th movie
			if(rsAll.absolute(5)) {
				System.out.println("\nAll movies:\nID\tMovie_Name\tGenre\tIMDB_Rating\tYear");
		        while (rsAll.next()) {
		            System.out.println(rsAll.getInt("ID") + "\t" +
		                    rsAll.getString("Movie_Name") + "\t" +
		                    rsAll.getString("Genre") + "\t\t" +
		                    rsAll.getFloat("IMDB_Rating") + "\t" +
		                    rsAll.getInt("Year"));
		        }
			}
			//iii. insert a new row
			System.out.print("Enter the id:");
			int id=sc.nextInt();
			System.out.print("Enter movie name:");
			String name=sc.next();
			System.out.print("Enter genre:");
			String genre=sc.next();
			System.out.print("Imdb rating:");
			float rating=sc.nextFloat();
			System.out.print("Enter year:");
			int year=sc.nextInt();
			
			String query3="INSERT INTO Movies (ID, Movie_Name, Genre, IMDB_Rating, Year) VALUES (?,?,?,?,?)";
			try(PreparedStatement pstmt=conn.prepareStatement(query3)){
				pstmt.setInt(1,id);
				pstmt.setString(2,name);
				pstmt.setString(3,genre);
				pstmt.setFloat(4,rating);
				pstmt.setInt(5,year);
				int rowsInserted=pstmt.executeUpdate();
				if(rowsInserted>0) {
					System.out.print("inserted new row");
				}
			}
			rsAll=stmt.executeQuery(queryAll);
			printResult(rsAll);
			
			//iv. delete a row 
			String delQuery="delete from movies where IMDB_Rating<5";
			int deletedRows=stmt.executeUpdate(delQuery);
			System.out.print("Rows deleted:"+ deletedRows);
			
			//v update
			String updateQuery="update movies set Genre='Sci-fi' where ID=10";
			int rowsUpdated=stmt.executeUpdate(updateQuery);
			System.out.print("Rows updated:"+ rowsUpdated);
			
			rsAll=stmt.executeQuery(queryAll);
			printResult(rsAll);
		}catch(SQLException e) {
			e.printStackTrace();
		}
	}
	private static void printResult(ResultSet rs) throws SQLException{
		System.out.println("\nAll movies:\nID\tMovie_Name\tGenre\tIMDB_Rating\tYear");
        while (rs.next()) {
            System.out.println(rs.getInt("ID") + "\t" +
                    rs.getString("Movie_Name") + "\t" +
                    rs.getString("Genre") + "\t\t" +
                    rs.getFloat("IMDB_Rating") + "\t" +
                    rs.getInt("Year"));
        }
	}
}
