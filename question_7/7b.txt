package mypack;

import java.sql.*;

public class BookDB_7b {
	private static final String URL="jdbc:mysql://localhost:3306/dbName",USER="root",PASS="";
	
	public static void main(String[] args) {
		try {
			Connection conn=DriverManager.getConnection(URL,USER,PASS);
			Statement stmt = conn.createStatement(ResultSet.TYPE_SCROLL_INSENSITIVE, ResultSet.CONCUR_UPDATABLE);
			
            String createDB="create database Books";
			stmt.executeUpdate(createDB);
			System.out.println("created DB");
			
			String createTable = "CREATE TABLE IF NOT EXISTS Book ("
                    + "Book_No INT AUTO_INCREMENT PRIMARY KEY, "
                    + "Title VARCHAR(255) NOT NULL, "
                    + "Author VARCHAR(255) NOT NULL, "
                    + "Publication VARCHAR(255) NOT NULL, "
                    + "Price DECIMAL(10,2) NOT NULL)";
            stmt.executeUpdate(createTable);
            System.out.println("Table created or already exists.");

            // Step 3: Insert Sample Data directly using Statement
            String insertData1 = "INSERT INTO Book (Title, Author, Publication, Price) VALUES "
                    + "('Java Programming', 'James Gosling', 'Sun Microsystems', 599.99)";
            String insertData2 = "INSERT INTO Book (Title, Author, Publication, Price) VALUES "
                    + "('Python Basics', 'Guido van Rossum', 'O’Reilly Media', 499.50)";
            String insertData3 = "INSERT INTO Book (Title, Author, Publication, Price) VALUES "
                    + "('Data Structures', 'Robert Lafore', 'Pearson', 699.00)";

            stmt.executeUpdate(insertData1);
            stmt.executeUpdate(insertData2);
            stmt.executeUpdate(insertData3);

            System.out.println("Data inserted successfully.");
            
		}catch (SQLException e) {
			e.printStackTrace();
		}
	}
}
