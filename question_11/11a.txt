package mypack;

import java.sql.*;

public class Q11_a {
	private static final String URL = "jdbc:mysql://localhost:3306/dbName", USER = "root", PASS = "";

    public static void main(String[] args) {
        Connection conn = null;
        Statement stmt = null;
        
        try {
            conn = DriverManager.getConnection(URL, USER, PASS);
            stmt = conn.createStatement();
            
            String createDB="create database Employee";
			stmt.executeUpdate(createDB);
			System.out.println("created DB");
			
            String createTableSQL = "CREATE TABLE IF NOT EXISTS Employee_11a (" +
                    "name VARCHAR(20), " +
                    "designation VARCHAR(20), " +
                    "dept VARCHAR(20), " +
                    "salary INT)";
            stmt.executeUpdate(createTableSQL);
            System.out.println("Table 'Employee' created or already exists.");

            String insertDataSQL = "INSERT INTO Employee_11a (name, designation, dept, salary) VALUES " +
                    "('Anika', 'Manager', 'Sales', 75000), " +
                    "('Rahul', 'Manager', 'HR', 65000), " +
                    "('Priya', 'Executive', 'Marketing', 45000), " +
                    "('Rohit', 'SDE', 'IT', 85000)" ;
            stmt.executeUpdate(insertDataSQL);
            System.out.println("Sample data inserted into 'Employee' table.");

            String querySQL = "SELECT * FROM Employee_11a WHERE salary > 50000";
            ResultSet rs = stmt.executeQuery(querySQL);

            System.out.println("\nEmployees with salary above Rs. 50,000:");
            System.out.println("Name\tDesignation\tDept\tSalary");
            while (rs.next()) {
                System.out.println(rs.getString("name") + "\t" +
                        rs.getString("designation") + "\t\t" +
                        rs.getString("dept") + "\t" +
                        rs.getInt("salary"));
            }
        } catch (SQLException e) {
            e.printStackTrace();
        }
    }
}
