import java.io.IOException;
import javax.servlet.ServletException;
import javax.servlet.annotation.WebServlet;
import javax.servlet.http.HttpServlet;
import javax.servlet.http.HttpServletRequest;
import javax.servlet.http.HttpServletResponse;

import java.sql.*;
import java.io.*;
/**
 * Servlet implementation class Subjects
 */
@WebServlet("/Subjects")
public class Subjects extends HttpServlet {
	private static final long serialVersionUID = 1L;
       
	 
	protected void doPost(HttpServletRequest request, HttpServletResponse response) throws ServletException, IOException {
		response.setContentType("text/html");
		PrintWriter out = response.getWriter();
		final String URL = "jdbc:mysql://localhost:3306/FacultyInfo", USER = "root", PASS = "";
		
		String facultyID = request.getParameter("facultyId");
		String subName = request.getParameter("subName");
		try {
			Class.forName("com.mysql.cj.jdbc.Driver");
			Connection conn = DriverManager.getConnection(URL,USER,PASS);
			Statement stmt = conn.createStatement();
			
			String query1="update Faculty set subName=? where facultyID=?";
			PreparedStatement pstmt = conn.prepareStatement(query1);
			pstmt.setString(1, subName);
			pstmt.setInt(2, Integer.parseInt(facultyID));
			int rowsUpdated=pstmt.executeUpdate();
			
			out.println("<h2>Subjects alloted to facultyID "+facultyID+"</h2>");
			
			String query2="select * from Faculty where facultyID=?";
			pstmt=conn.prepareStatement(query2);
			pstmt.setInt(1, Integer.parseInt(facultyID));
			ResultSet rs = pstmt.executeQuery();
			
			out.println("<table border='1'><tr><th>SubID</th><th>SubName</th><th>FacultyID</th></tr>");
			while(rs.next()) {
				out.println("<tr><td>"+rs.getInt("subID")+"</td><td>"+rs.getString("subName")+"</td><td>"+rs.getInt("facultyID")+"</td></tr>");
			}
			out.println("</table>");
			
		}catch(Exception e) {
			e.printStackTrace();
		}
	}

}
